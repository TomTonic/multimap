// goClasses are the Go malloc size classes up to 8 KiB.
var goClasses = []int{8, 16, 24, 32, 48, 64, 80, 96, 112, 128, 144, 160, 176, 192, 208, 224, 240, 256, 288, 320, 352, 384, 416, 448, 480, 512, 576, 640, 704, 768, 896, 1024, 1152, 1280, 1408, 1536, 1792, 2048, 2304, 2688, 3072, 3200, 3456, 4096, 4864, 5376, 6144, 6528, 6784, 6912, 8192}

// Block returns the size class an object of size occupies, and the offset of the
// object in its block (8 with a malloc header).
func Block(size int, ptrs bool) (block, off int) {
	n := size
	if ptrs && size > 512 {
		n += 8
		off = 8
	}
	for _, c := range goClasses {
		if c >= n {
			return c, off
		}
	}
	return n, off
}

// Objects calls fn for every object of the tree: a label, its size and whether it holds pointers.
func (m *Map[T]) Objects(fn func(label string, size int, ptrs bool)) {
	if m.t.root != nil {
		m.objWalk(m.t.root, fn)
	}
}

var kindNames = map[kind]string{kN5: "N5", kN12: "N12", kN26: "N26", kN58: "N58", kN256: "N256", kR8: "R8", kR24: "R24", kR56: "R56", kR256: "R256"}

func (m *Map[T]) leafSize(l *leafHead) (string, int, bool) {
	var z T
	switch {
	case l.cls() > 0 && m.flat == 1:
		return "flat leaf", int(flatSizes[l.cls()]), false
	case l.cls() > 0 && m.flat == 2:
		return "typed leaf", int(typedOff(int(l.klen))) + typedCaps[l.cls()]*int(unsafe.Sizeof(z)), true
	}
	var s uintptr
	switch k := l.klen; {
	case k <= 16:
		s = unsafe.Sizeof(leaf[T, [16]byte]{})
	case k <= 32:
		s = unsafe.Sizeof(leaf[T, [32]byte]{})
	case k <= 48:
		s = unsafe.Sizeof(leaf[T, [48]byte]{})
	case k <= 64:
		s = unsafe.Sizeof(leaf[T, [64]byte]{})
	case k <= 96:
		s = unsafe.Sizeof(leaf[T, [96]byte]{})
	case k <= 128:
		s = unsafe.Sizeof(leaf[T, [128]byte]{})
	case k <= 192:
		s = unsafe.Sizeof(leaf[T, [192]byte]{})
	case k <= maxInline:
		s = unsafe.Sizeof(leaf[T, [256]byte]{})
	default:
		s = unsafe.Sizeof(leaf[T, string]{})
	}
	return "set leaf", int(s), true
}

func (m *Map[T]) objWalk(n *header, fn func(string, int, bool)) {
	switch {
	case isLeaf(n.kind):
		a, b, c := m.leafSize(asLeaf(n))
		fn(a, b, c)
		return
	case isPage(n.kind):
		fn("page", 64<<asPage(n).class, false)
		return
	}
	sz := int(fixedSize[n.kind&kindMask])
	switch tailClass(n.pathLen()) {
	case tail16, tailStr:
		sz += 16
	case tail48:
		sz += 48
	case tail112:
		sz += 112
	}
	lbl := kindNames[n.kind]
	if tailClass(n.pathLen()) != tailNone {
		lbl += "+tail"
	}
	fn(lbl, sz, true)
	if t := termOf(n); t != nil {
		a, b, c := m.leafSize(t)
		fn(a, b, c)
	}
	if isRange(n.kind) {
		for _, c := range asR(n).children()[:asR(n).n] {
			m.objWalk(c, fn)
		}
		return
	}
	eachInner(n, func(_ byte, c *header) { m.objWalk(c, fn) })
}
