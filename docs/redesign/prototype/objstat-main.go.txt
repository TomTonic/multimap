// Command objstat is TEMPORARY: it counts, per benchmark case, how many objects of the
// ordered multimap miss the cache-line criterion.
package main

import (
	"fmt"
	"os"
	"sort"
	"strconv"

	"github.com/TomTonic/multimap/bench/keys"
	"github.com/TomTonic/multimap/internal/art"
)

type acc struct {
	n, bad64, bad128, extra int
	bytes, wasted           int
	by                      map[string][3]int // label -> count, bad64, bad128
}

func (a *acc) add(label string, size int, ptrs bool) {
	block, off := art.Block(size, ptrs)
	a.n++
	a.bytes += block
	a.wasted += block - size - off
	b64, b128 := block%64 != 0, block%128 != 0
	if b64 {
		a.bad64++
	}
	if b128 {
		a.bad128++
	}
	// lines touched beyond the minimum, averaged over the alignments the block has
	minLines := (size + 63) / 64
	ex := 0
	for k := range 64 {
		s := (k*block + off) % 64
		if (s+size-1)/64+1 > minLines {
			ex++
		}
	}
	a.extra += ex // sixty-fourths of an object
	v := a.by[label]
	v[0]++
	if b64 {
		v[1]++
	}
	a.by[label] = v
}

func build[T comparable](kind keys.Kind, n int, unique bool, val func(uint64) T) *acc {
	c := keys.Generate(kind, n, 0xC0FFEE)
	vals, offs := keys.Values(n, 0xFA11)
	if c.Natural != nil {
		vals, offs = nil, make([]int, n+1)
		for i, vs := range c.Natural[:n] {
			vals = append(vals, vs...)
			offs[i+1] = len(vals)
		}
	}
	var m art.Map[T]
	for i, key := range c.Keys.B {
		vs := vals[offs[i]:offs[i+1]]
		if unique {
			vs = vs[:1]
		}
		for _, v := range vs {
			m.Add(key, val(v))
		}
	}
	a := &acc{by: map[string][3]int{}}
	m.Objects(a.add)
	return a
}

func main() {
	kinds := []keys.Kind{keys.U64, keys.Str, keys.UUID, keys.Email, keys.URL, keys.Path, keys.Street}
	sizes := []int{4096, 16384, 262144, 1048576}
	detail := len(os.Args) > 1 && os.Args[1] == "detail"
	fmt.Println("| Case | Objekte | kein Vielfaches von 64 | kein Vielfaches von 128 | mehr Zeilen als nötig | Anteil Blatt/Knoten/Seite |")
	fmt.Println("|---|---:|---:|---:|---:|---|")
	for _, kind := range kinds {
		for _, prof := range []string{"multi", "unique", "multi-str", "unique-str"} {
			for _, n := range sizes {
				if n > keys.Capacity(kind) {
					continue
				}
				unique := prof == "unique" || prof == "unique-str"
				var a *acc
				if prof == "multi" || prof == "unique" {
					a = build(kind, n, unique, func(v uint64) uint64 { return v })
				} else {
					a = build(kind, n, unique, func(v uint64) string { return fmt.Sprintf("%016x", v) })
				}
				pct := func(x int) string { return strconv.FormatFloat(100*float64(x)/float64(a.n), 'f', 1, 64) + " %" }
				frac := strconv.FormatFloat(100*float64(a.extra)/64/float64(a.n), 'f', 1, 64) + " %"
				var parts []string
				for l, v := range a.by {
					parts = append(parts, fmt.Sprintf("%s %.0f%%", l, 100*float64(v[0])/float64(a.n)))
				}
				sort.Strings(parts)
				mix := ""
				for _, p := range parts {
					mix += p + "; "
				}
				fmt.Printf("| %s %s %s | %d | %s | %s | %s | %s |\n", kind, prof, human(n), a.n, pct(a.bad64), pct(a.bad128), frac, mix)
				_ = detail
			}
		}
	}
}

func human(n int) string {
	switch n {
	case 1048576:
		return "1M"
	case 262144:
		return "256K"
	}
	return strconv.Itoa(n/1024) + "K"
}
